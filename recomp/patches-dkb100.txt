set hidproxy DKB_SKIP_HID_PROXY 1

# al::GameFrameworkOunce::createControllerMgr (main+0x73b300) reaches sub_71008725E0 through main+0x73be90: it makes
# a transfer memory and talks to IHidServer itself (nn::hid::server::CreateHidServerProxy), past the nn::hid calls the
# host takes over, and aborts on a failed Result. The host serves no "hid" IPC service, so the call reports success.
site hidproxy main 73bea0 9404d9d0 2a1f03e0   # bl sub_71008725E0 -> mov w0, wzr

# EXPERIMENT, after Eglantine-ZendNevs v2.0.5 (GPL-3.0, src/core/loader/nso.cpp): the game's clock follows the real frame
# time down to 24 fps. DKB_DYNAMIC_FPS=0 turns it off. Written without its code cave: replacements may not branch.
set dynfps DKB_DYNAMIC_FPS 1
# The frame-rate controller reads the whole frame, [fw+0xe8], instead of the calc time, [fw+0x128].
site dynfps main 1ea930 f9409429 f9407429
# It keeps the frame time in 60ths in an unused field: str s0, [x19, #0x5c]
site dynfps main 1ea968 2944266d bd005e60 2944266d
# Sleep below 40 fps too: the check's flag loads as zero, so the cbnz after it never leaves.
site dynfps main 1ea5d4 39472e68 2a1f03e8
# Take the game's own sleep out of the frame time: frame start += max(sleep, 0). subs stays the only flag-using word.
site dynfps main 1ea610 eb080120 eb080120 8aa0fc09 f9407a6a 8b09014a f9007a6a
# The clock's step: the longer of 60/target and the frame time, at most 2.5 (24 fps).
site dynfps main 2d9fb0 52a84e08 bd405d02 52a84e08
site dynfps main 2d9fc8 1e201820 1e201820 1e226800 1e209001 1e217800
